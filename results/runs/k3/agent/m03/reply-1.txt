```ts
import { expect, test } from '@playwright/test';

/** Locator strategy: accessible roles and visible text. */
test('marks a task as complete and updates the counter', async ({ page }) => {
  await page.goto('/');

  await page.getByRole('checkbox', { name: 'Toggle Write the reproduction guide' }).check();

  await expect(page.locator('#items-left')).toHaveText('1 item left');
  await expect(
    page.getByRole('listitem').filter({ hasText: 'Write the reproduction guide' }),
  ).toHaveClass(/is-completed/);
});
```
